    74→function formatPrice(price: number): string {
    75→  if (price === 0) return "تماس بگیرید";
    76→  return new Intl.NumberFormat("fa-IR").format(price) + " تومان";
    77→}
    78→
    79→function faNum(n: number | string): string {
    80→  return String(n).replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[parseInt(d)]);
    81→}
    82→
    83→function parsePower(power: string): number {
    84→  const match = power.match(/([\d.]+)\s*(kw|HP)/i);
    85→  if (!match) return 0;
    86→  const val = parseFloat(match[1]);
    87→  return match[2].toLowerCase() === "kw" ? val : val * 0.746;
    88→}
    89→
    90→function powerLabel(power: string): string {
    91→  return power;
