    83→/* ─────────────────────────── Helpers ─────────────────────────── */
    84→function formatPrice(price: number): string {
    85→  if (price === 0) return "تماس بگیرید";
    86→  return new Intl.NumberFormat("fa-IR").format(price) + " تومان";
    87→}
    88→
    89→function faNum(n: number | string): string {
    90→  return String(n).replace(/\d/g, (d) =>
    91→    "\u06F0\u06F1\u06F2\u06F3\u06F4\u06F5\u06F6\u06F7\u06F8\u06F9"[
    92→      parseInt(d)
    93→    ]
    94→  );
    95→}
