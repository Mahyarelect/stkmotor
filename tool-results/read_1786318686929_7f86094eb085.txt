    82→function formatPrice(price: number): string {
    83→  if (price === 0) return "تماس بگیرید";
    84→  return new Intl.NumberFormat("fa-IR").format(price) + " تومان";

    87→function faNum(n: number | string): string {
    88→  return String(n).replace(/\d/g, (d) => "\u06F0\u06F1\u06F2\u06F3\u06F4\u06F5\u06F6\u06F7\u06F8\u06F9"[parseInt(d)]);
    89→}

   640→function FamilyCard({ family }: { family: ProductFamily }) {
   641→  // Get unique sizes sorted numerically
   642→  const uniqueSizes = [...new Set(family.variants.map((v) => v.size).filter(Boolean))].sort(
   643→    (a, b) => parseInt(a) - parseInt(b)
   644→  );
   645→
   646→  // Default: select first variant
   647→  const firstInStock = family.variants.find((v) => v.inStock) || family.variants[0];
   648→  const [selectedSize, setSelectedSize] = useState(firstInStock?.size || "");
   649→
   650→  const variantsForSize = family.variants.filter((v) => v.size === selectedSize);
   651→  const activeVariant = variantsForSize[0] || family.variants[0];
   652→
   653→  // Price range
   654→  const prices = family.variants.filter((v) => v.price > 0).map((v) => v.price);
   655→  const minPrice = prices.length > 0 ? Math.min(...prices) : 0;
   656→  const maxPrice = prices.length > 0 ? Math.max(...prices) : 0;
   657→
   658→  return (
   659→    <Card className="overflow-hidden border-gray-200 hover:shadow-lg hover:border-blue-200 transition-all group flex flex-col">
   660→      {/* Image Placeholder */}
   661→      <div className="relative bg-gradient-to-br from-gray-50 to-gray-100 h-44 flex items-center justify-center overflow-hidden">
   662→        <div className="text-center">
   663→          <Cog size={40} className="text-gray-300 mx-auto mb-1 group-hover:scale-110 transition-transform" />
   664→          <p className="text-gray-400/60 text-xs">تصویر محصول</p>
   665→        </div>
   666→        {/* Category badge */}
   667→        <Badge
   668→          className={`absolute top-3 right-3 text-[10px] px-2 py-0.5 ${
   669→            family.category === "single-phase"
   670→              ? "bg-amber-500 text-white"
   671→              : "bg-blue-600 text-white"
   672→          }`}
   673→        >
   674→          {family.phase}
   675→        </Badge>
   676→      </div>
   677→
   678→      <CardContent className="p-5 flex flex-col flex-1">
   679→        {/* Name */}
   680→        <Link href={`/product/${family.slug}`} className="block">
   681→          <h4 className="font-bold text-gray-900 text-base mb-1 group-hover:text-blue-700 transition-colors">
   682→            الکتروموتور {family.phase} STK
   683→          </h4>
   684→        </Link>
   685→        <p className="text-xs text-gray-400 mb-3">
   686→          پوسته {family.shellType} · {family.speed ? `${faNum(family.speed)} دور` : ""}
   687→        </p>
   688→
   689→        {/* Size Selector Chips */}
   690→        {uniqueSizes.length > 1 && (
   691→          <div className="mb-3">
   692→            <p className="text-[10px] text-gray-500 mb-1.5 font-medium">سایز فریم</p>
   693→            <div className="flex flex-wrap gap-1.5">
   694→              {uniqueSizes.map((size) => {
   695→                const hasVariant = family.variants.some((v) => v.size === size);
   696→                const hasInStock = family.variants.some((v) => v.size === size && v.inStock);
   697→                return (
   698→                  <button
   699→                    key={size}
   700→                    onClick={() => hasVariant && setSelectedSize(size)}
   701→                    disabled={!hasVariant}
   702→                    className={`px-2.5 py-1 rounded-md text-xs font-medium num-en transition-all ${
   703→                      selectedSize === size
   704→                        ? "bg-blue-700 text-white shadow-sm"
   705→                        : hasInStock
   706→                          ? "bg-gray-100 text-gray-700 hover:bg-blue-50 hover:text-blue-700"
   707→                          : "bg-gray-50 text-gray-300 line-through cursor-not-allowed"
   708→                    }`}
   709→                  >
   710→                    {size}
   711→                  </button>
   712→                );
   713→              })}
   714→            </div>
   715→          </div>
   716→        )}
   717→
   718→        {/* Spec Grid (3 items) */}
   719→        <div className="grid grid-cols-3 gap-2 mb-4">
   720→          <SpecItem
   721→            label="توان"
   722→            value={activeVariant?.power || "-"}
   723→          />
   724→          <SpecItem
   725→            label="دور"
   726→            value={activeVariant?.speed ? `${faNum(activeVariant.speed)} RPM` : "-"}

   731→          />
   732→        </div>

   734→        {/* Stock Status */}
   735→        <div className="mb-4">

   759→            <div className="text-sm text-gray-500 num-en">

   763→            <div className="text-sm text-gray-500">تماس بگیرید</div>

   772→        </Link>
   773→      </CardContent>
   774→    </Card>
   775→  );
